1. Inserting .nwb files using `insert_sessions`. If the probe description in 
   the nwb file is different from one already inserted, it will throw an error
   saying that the probe name matches but the description does not.
   Used the follwoing, for example, to fix this:
   ```
   import h5py
   
   nwb_path = "/media/labuser/NA_1_2025/spyglass/Senor/Senor_20201220/raw/Senor20201220.nwb"
   
   old_desc = "128 channel polyimide probe"
   new_desc_6mm = "128 channel polyimide 4s6mm-15um-26um"
   new_desc_8mm = "128 channel polyimide 4s8mm-20um-40um"
   
   with h5py.File(nwb_path, "r+") as f:
       devs = f["/general/devices"]
       for name, grp in devs.items():
           probe_type = grp.attrs.get("probe_type", None)
           desc = grp.attrs.get("probe_description", None)
   
           # normalize to str for comparison
           if isinstance(desc, bytes):
               desc_str = desc.decode("utf-8")
           else:
               desc_str = desc
   
           if desc_str != old_desc:
               continue
   
           if isinstance(probe_type, bytes):
               probe_type_str = probe_type.decode("utf-8")
           else:
               probe_type_str = probe_type
   
           if probe_type_str == "128c-4s8mm6cm-20um-40um-sl":
               print(f"{name}: {desc_str!r} -> {new_desc_8mm!r}")
               grp.attrs["probe_description"] = new_desc_8mm
           elif probe_type_str == "128c-4s6mm6cm-15um-26um-sl":
               print(f"{name}: {desc_str!r} -> {new_desc_6mm!r}")
               grp.attrs["probe_description"] = new_desc_6mm
   
   from pynwb import NWBHDF5IO
   
   with NWBHDF5IO(nwb_path, mode="r", load_namespaces=True) as io:
       nwbf = io.read()
       for dev_name, dev in nwbf.devices.items():
           if hasattr(dev, "probe_type") and hasattr(dev, "probe_description"):
               print(dev_name)
               print("  probe_type:        ", dev.probe_type)
               print("  probe_description: ", dev.probe_description)
   
   ```
